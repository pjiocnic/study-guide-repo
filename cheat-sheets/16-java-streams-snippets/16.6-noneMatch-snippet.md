import java.time.LocalDate;
import java.util.Arrays;
import java.util.List;

public class Main {

    public static void main(String[] args) {
        // Create a list of courses
        List<Course> courses = Arrays.asList(
                new Course("Math", LocalDate.of(2024, 1, 10), LocalDate.of(2024, 6, 20)),
                new Course("History", LocalDate.of(2024, 3, 1), LocalDate.of(2024, 8, 15)),
                new Course("Science", LocalDate.of(2024, 6, 1), null) // This course has a null end date
        );

        // Use noneMatch to check if no course has a null end date
        boolean allCoursesHaveEndDate = courses.stream()
                .noneMatch(course -> course.getEndDate() == null);

        // Output the result
        if (allCoursesHaveEndDate) {
            System.out.println("All courses have an end date.");
        } else {
            System.out.println("Some courses are missing an end date.");
        }
    }
}
