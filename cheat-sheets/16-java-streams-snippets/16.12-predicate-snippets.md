### Combining filters / predicates using and

```java
import java.util.function.Predicate;
import java.util.stream.Stream;
import java.util.List;
import java.util.stream.Collectors;

class App {
    public static void main(String[] args) {
        Predicate<String> startsWithJ = (word) -> word.startsWith("j");
        Predicate<String> fourChars = (word) -> word.length() == 4;

        var words = Stream.of("java", "javascript", "jakarta")
                          .filter(startsWithJ)
                          .filter(fourChars)
                          .collect(Collectors.toList());

        System.out.println(words);
    }
}
```

### Stream filtering with composed / Reused Predicate Methods

```java
import java.util.List;
import java.util.function.Predicate;
import java.util.stream.Collectors;
import java.util.stream.Stream;

class App {
    public static void main(String[] args) {
        var languages = Stream.of("java", "js", "c", "vb", "ada");

        Predicate<String> startsWithJ = App::startsWithJ;
        Predicate<String> brief = App::brief;
        Predicate<String> containsA = App::containsA;

        var composed = startsWithJ.and(brief).or(containsA);

        List<String> filtered = languages
                                  .filter(composed)
                                  .collect(Collectors.toList());
        System.out.println(filtered);
    }

    static boolean startsWithJ(String s) {
        return s.startsWith("j");
    }

    static boolean brief(String s) {
        return s.length() <= 3;
    }

    static boolean containsA(String s) {
        return s.contains("a");
    }
}
```
