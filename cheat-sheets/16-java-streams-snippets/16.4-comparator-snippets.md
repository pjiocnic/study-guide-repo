import lombok.AllArgsConstructor;
import lombok.EqualsAndHashCode;
import java.time.LocalDateTime;
import java.util.Comparator;
import java.util.Set;
import java.util.TreeSet;

@AllArgsConstructor
@EqualsAndHashCode(onlyExplicitlyIncluded = true)
class Building {

  @EqualsAndHashCode.Include
  public Address address;
  private LocalDateTime startDttm;

  public LocalDateTime getStartDttm() {
    return startDttm;
  }
}

@AllArgsConstructor
@EqualsAndHashCode
class Address {
  public String id;
  public AddressType type;
}

enum AddressType {
   COMMERCIAL("COMMERCIAL"),
   RESIDENCE("RESIDENCE");

   private final String type;
   AddressType(String type) {
     this.type = type;
   }
}

public class Main {
  public static void main(String[] args) {
    // Comparator that first compares by Address, then by startDttm in descending order
    Comparator<Building> buildingComparator = Comparator
        .comparing((Building b) -> b.address)
        .thenComparing(Building::getStartDttm, Comparator.reverseOrder());

    Set<Building> buildings = new TreeSet<>(buildingComparator);

    Address addr1 = new Address("1", AddressType.COMMERCIAL);
    Building blg1 = new Building(addr1, LocalDateTime.now());
    buildings.add(blg1);

    // Same Address, different startDttm
    Address addr2 = new Address("1", AddressType.COMMERCIAL);
    Building blg2 = new Building(addr2, LocalDateTime.now().plusDays(1));
    buildings.add(blg2);

    // Print the set to observe the order and unique entries
    System.out.println("Number of buildings: " + buildings.size()); // Should print 1
    buildings.forEach(b -> System.out.println(b.getStartDttm() + " - " + b.address.id + " - " + b.address.type));
  }
}
